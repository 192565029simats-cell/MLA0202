import numpy as np
import matplotlib.pyplot as plt

from sklearn.linear_model import LinearRegression
from sklearn.model_selection import train_test_split
from sklearn.metrics import mean_squared_error, r2_score

# Generate sales dataset
np.random.seed(42)

months = np.arange(1, 51).reshape(-1, 1)

sales = 1000 + 50 * months.ravel() + np.random.normal(0, 100, 50)

# Split dataset
X_train, X_test, y_train, y_test = train_test_split(
    months, sales, test_size=0.2, random_state=42
)

# Train model
model = LinearRegression()
model.fit(X_train, y_train)

# Predict sales
y_pred = model.predict(X_test)

print("Future Sales Prediction")
print("MSE:", mean_squared_error(y_test, y_pred))
print("R2 Score:", r2_score(y_test, y_pred))

# Predict future months
future_months = np.arange(51, 61).reshape(-1, 1)
future_sales = model.predict(future_months)

print("\nFuture Sales Predictions:")
for month, sale in zip(future_months.ravel(), future_sales):
    print("Month", month, "Sales:", round(sale, 2))

# Graph 1: Actual sales and prediction line
plt.figure(figsize=(9, 5))
plt.scatter(months, sales, label="Actual Sales")
plt.plot(months, model.predict(months),
         color="red", label="Prediction Line")
plt.xlabel("Month")
plt.ylabel("Sales")
plt.title("Sales Prediction")
plt.legend()
plt.grid()
plt.show()

# Graph 2: Future sales
plt.figure(figsize=(8, 5))
plt.plot(future_months, future_sales,
         marker="o", label="Future Sales")
plt.xlabel("Future Month")
plt.ylabel("Predicted Sales")
plt.title("Future Sales Forecast")
plt.legend()
plt.grid()
plt.show()
